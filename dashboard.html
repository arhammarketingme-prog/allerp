// 🌐 बारकोड / QR डेटा हँडलर (Multi-Field Smart Name Fallback) async function handleBarcodeDirectLookup(barcodeVal) { if (!barcodeVal || barcodeVal.trim().length < 3) return; let raw = barcodeVal.trim(); const statusEl = document.getElementById('api-status-indicator'); if (statusEl) statusEl.innerHTML = ' प्रॉडक्ट शोधत आहे...'; // GS1 कोड क्लीनिंग let cleanCode = raw; if (raw.startsWith('01') && raw.length >= 16) { cleanCode = raw.substring(2, 16); } else if (raw.includes('890')) { const match = raw.match(/890\d{9,11}/); if (match) cleanCode = match[0]; } try { // १. आधी Supabase सेंट्रल कॅटलॉग तपासणे const { data: dbItem } = await sb.from('global_catalog') .select('*') .or(`barcode.eq.${raw},barcode.eq.${cleanCode}`) .maybeSingle(); if (dbItem && dbItem.name) { document.getElementById('np-name').value = dbItem.name; document.getElementById('np-unit').value = dbItem.unit || 'packet'; if (dbItem.selling_price) document.getElementById('np-price').value = dbItem.selling_price; setCentralPreview({ name: dbItem.name, cat: dbItem.category || 'Central Catalog', img: dbItem.image_url || generateCategoryVector(dbItem.name, 'Spices'), unit: dbItem.unit }); if (statusEl) statusEl.textContent = '✅ प्रॉडक्ट लोड झाले!'; return; } // २. थेट OpenFoodFacts जागतिक API तपासणे const res = await fetch(`https://world.openfoodfacts.org/api/v0/product/${cleanCode}.json`); const json = await res.json(); if (json.status === 1 && json.product) { const p = json.product; // 🌟 मल्टी-फील्ड स्मार्ट नेम एक्सट्रॅक्टर (नाव कधीही रिकामे राहणार नाही) let resolvedName = p.product_name || p.product_name_en || p.generic_name || p.generic_name_en || p._keywords?.slice(0, 3)?.join(' ') || ''; const brandName = p.brands || p.brand_owner || ''; const qty = p.quantity ? ` ${p.quantity}` : ''; // जर नाव मिळाले नाही तर ब्रँड + कॅटेगरी वापरणे if (!resolvedName && brandName) { resolvedName = `${brandName} Spices/Masala`; } else if (resolvedName && brandName && !resolvedName.toLowerCase().includes(brandName.toLowerCase())) { resolvedName = `${brandName} ${resolvedName}`; } const finalFullName = (resolvedName ? resolvedName + qty : "Cook With Masala Special Pack").trim(); const realImg = p.image_front_url || p.image_front_small_url || p.image_url; let unit = 'packet'; if (p.quantity && (p.quantity.includes('g') || p.quantity.includes('kg'))) unit = 'kg'; // इनपुट बॉक्समध्ये थेट नाव भरणे document.getElementById('np-name').value = finalFullName; document.getElementById('np-unit').value = unit; setCentralPreview({ name: finalFullName, cat: brandName || 'मसाले व पॅकेज्ड फूड', img: realImg || generateCategoryVector(finalFullName, 'Spices'), unit: unit }); if (statusEl) statusEl.textContent = '✅ उत्पादनाचे नाव व फोटो लोड झाले!'; } else { // जर डेटाबेसमध्ये सापडले नाही, तर डीफॉल्ट नाव तयार करणे const fallbackName = "Cook With Masala Pack"; document.getElementById('np-name').value = fallbackName; setCentralPreview({ name: fallbackName, cat: "मसाले (Spices)", img: generateCategoryVector(fallbackName, 'Spices'), unit: "packet" }); if (statusEl) statusEl.textContent = 'नवीन बारकोड नोंदवला. नाव चेक करा.'; } } catch (err) { if (statusEl) statusEl.textContent = 'स्थानिक मोड सक्रिय'; } }
