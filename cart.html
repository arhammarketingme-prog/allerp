// 🚀 सुधारित स्प्लिट ऑर्डर सबमिट फंक्शन (Multi-Vendor Split Order & Direct Insert) async function submitCartOrder(event) { event.preventDefault(); const submitBtn = document.getElementById('cart-submit-order-btn'); if (submitBtn && submitBtn.disabled) return; if (submitBtn) { submitBtn.disabled = true; submitBtn.textContent = '⏳ ऑर्डर पाठवत आहे...'; } const name = document.getElementById('cartCustName').value.trim(); const phone = document.getElementById('cartCustPhone').value.trim(); const address = document.getElementById('cartCustAddress').value.trim(); const paymentMode = document.getElementById('selectedPaymentMode').value; const cart = typeof getCart === 'function' ? getCart() : []; if (!cart || cart.length === 0) { alert('तुमची कार्ट रिकामी आहे!'); if (submitBtn) { submitBtn.disabled = false; submitBtn.textContent = '🔒 सुरक्षित ऑर्डर सबमिट करा'; } return; } try { const { data: { user: currentUser } } = await sb.auth.getUser(); if (!currentUser) { alert('⚠️ तुमचे सत्र संपले आहे किंवा तुम्ही लॉग-इन नाही. कृपया पुन्हा लॉगिन करा.'); window.location.href = 'login.html'; return; } // १. वेगवेगळ्या दुकानांनुसार प्रॉडक्ट्सचे ग्रुप्स पाडणे (Split Orders by Business ID) let merchantGroups = {}; cart.forEach(item => { let bId = item.business_id; if (!bId) return; if (!merchantGroups[bId]) { merchantGroups[bId] = { business_id: bId, items: [] }; } merchantGroups[bId].items.push(item); }); let groupKeys = Object.keys(merchantGroups); if (groupKeys.length === 0) { alert('त्रुटी: कार्टमधील प्रॉडक्ट्सशी कोणतेही वैध दुकान जोडलेले नाही.'); if (submitBtn) { submitBtn.disabled = false; submitBtn.textContent = '🔒 सुरक्षित ऑर्डर सबमिट करा'; } return; } let firstMerchantHandle = 'store'; let firstMerchantEmail = ''; // २. प्रत्येक दुकानदारासाठी स्वतंत्र ऑर्डर रेकॉर्ड सेव्ह करणे for (let bId of groupKeys) { let group = merchantGroups[bId]; let itemsSummaryText = group.items.map(i => `${i.name} (×${i.quantity})`).join(', '); let totalAmt = group.items.reduce((sum, i) => sum + (Number(i.price) * Number(i.quantity)), 0); let orderOtp = Math.floor(1000 + Math.random() * 9000).toString(); let { data: bizData } = await sb.from('businesses') .select('id, owner_id, owner_username, owner_email, name, slug') .eq('id', bId) .maybeSingle(); let targetMerchantUserId = bizData ? bizData.owner_id : null; let targetHandle = bizData ? (bizData.owner_username || bizData.slug) : 'store'; let targetEmail = bizData ? bizData.owner_email : ''; if (groupKeys.indexOf(bId) === 0) { firstMerchantHandle = targetHandle; firstMerchantEmail = targetEmail; } // थेट orders टेबलमध्ये रेकॉर्ड टाकणे (कोणत्याही RPC फंक्शनची गरज नाही) const orderPayload = { business_id: bId, customer_name: name, customer_phone: phone, delivery_address: address, items_summary: itemsSummaryText, total_amount: totalAmt, payment_method: paymentMode, status: 'pending', customer_user_id: currentUser.id, delivery_pin: orderOtp }; const { error: insertErr } = await sb.from('orders').insert([orderPayload]); if (insertErr) { console.error('Order save error:', insertErr.message); throw new Error(insertErr.message); } // ३. संबंधित दुकानदाराच्या वीव्हो चॅटमध्ये मेसेज सिंक करणे if (targetMerchantUserId || targetEmail || targetHandle) { try { if (!targetMerchantUserId && targetEmail) { let { data: prof } = await sb.from('profiles').select('id').eq('email', targetEmail).maybeSingle(); if (prof) targetMerchantUserId = prof.id; } if (!targetMerchantUserId && targetHandle) { let { data: prof } = await sb.from('profiles').select('id').ilike('username', targetHandle).maybeSingle(); if (prof) targetMerchantUserId = prof.id; } if (targetMerchantUserId) { let targetConvId = null; let { data: myConvs } = await sb.from('conversation_members').select('conversation_id').eq('user_id', currentUser.id); let myIds = (myConvs || []).map(r => r.conversation_id); if (myIds.length) { let { data: theirConvs } = await sb.from('conversation_members').select('conversation_id').eq('user_id', targetMerchantUserId).in('conversation_id', myIds); if (theirConvs && theirConvs.length) targetConvId = theirConvs[0].conversation_id; } if (!targetConvId) { let { data: newConv } = await sb.from('conversations').insert({ type: 'direct', created_by: currentUser.id }).select().single(); if (newConv) { targetConvId = newConv.id; await sb.from('conversation_members').insert([ { conversation_id: targetConvId, user_id: currentUser.id }, { conversation_id: targetConvId, user_id: targetMerchantUserId } ]); } } if (targetConvId) { let shopNameTitle = bizData ? bizData.name : 'Store'; let orderMsgText = `📦 नवीन ऑनलाईन ऑर्डर (दुकान: ${shopNameTitle}):\n👤 ग्राहक: ${name}\n📱 मोबाईल: ${phone}\n🏠 पत्ता: ${address}\n🛒 तपशील: ${itemsSummaryText}\n💰 एकूण: ₹${totalAmt}\n🔐 OTP: ${orderOtp}`; await sb.from('messages').insert({ conversation_id: targetConvId, sender_id: currentUser.id, content: orderMsgText }); } } } catch (bridgeErr) { console.log("Weavo split sync note:", bridgeErr); } } } if (typeof saveCart === 'function') { saveCart([]); } else { localStorage.removeItem('cart'); localStorage.removeItem('marketplace_cart'); } closeCartModal(); alert(`✅ सर्व दुकानांच्या ऑर्डर्स यशस्वीरीत्या नोंदवल्या गेल्या!`); var orderMessage = '📦 **AllERP युनिफाइड ऑर्डर (Split)**\n\n' + '👤 **ग्राहक:** ' + name + '\n' + '📱 **मोबाईल:** ' + phone + '\n' + '📍 **पत्ता:** ' + address + '\n' + '💳 **पेमेंट:** ' + paymentMode; var weavoUrl = 'https://arhammarketingme-prog.github.io/weavo/?store=' + encodeURIComponent(firstMerchantHandle) + (firstMerchantEmail ? ('&merchant_email=' + encodeURIComponent(firstMerchantEmail)) : '') + '&prefill_msg=' + encodeURIComponent(orderMessage) + '&customer_email=' + encodeURIComponent(currentUser?.email || ''); window.open(weavoUrl, '_blank'); window.location.href = 'index.html'; } catch (err) { console.log('Database order error:', err); alert('त्रुटी: ' + err.message); if (submitBtn) { submitBtn.disabled = false; submitBtn.textContent = '🔒 सुरक्षित ऑर्डर सबमिट करा'; } } }
